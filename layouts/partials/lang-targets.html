{{- /* Returns a slice of dicts {lang, name, url, permalink, current} - one per configured language, resolving to this page's translation when one exists, else that language's home page. Used by the header language switcher and the hreflang tags in head.html. */ -}} {{- $page := . -}} {{- $result := slice -}} {{- range $page.Sites -}} {{- $lng := .Language.Lang -}} {{- $current := eq $lng $page.Language.Lang -}} {{- $target := .Home -}} {{- if $current -}} {{- $target = $page -}} {{- else -}} {{- range $page.Translations -}} {{- if eq .Language.Lang $lng }}{{ $target = . }}{{ end -}} {{- end -}} {{- end -}} {{- $result = $result | append (dict "lang" $lng "name" .Language.LanguageName "url" $target.RelPermalink "permalink" $target.Permalink "current" $current) -}} {{- end -}} {{- return $result -}}
